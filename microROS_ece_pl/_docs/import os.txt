import os
from launch import LaunchDescription
from launch_ros.actions import Node
from launch_ros.substitutions import FindPackageShare

def generate_launch_description():
    pkg_share = FindPackageShare('wave_rover_navigation').find('wave_rover_navigation')
    
    urdf_file = os.path.join(pkg_share, 'urdf', 'wave_rover.urdf')
    with open(urdf_file, 'r') as infp:
        robot_desc = infp.read()

    return LaunchDescription([
        # 1. Le Corps du Robot
        Node(
            package='robot_state_publisher',
            executable='robot_state_publisher',
            name='robot_state_publisher',
            output='screen',
            parameters=[{'robot_description': robot_desc}],
        ),

        # TFs statiques (Fixation du Lidar et de l'IMU)
        Node(
            package='tf2_ros',
            executable='static_transform_publisher',
            name='base_to_laser',
            # ---> CORRECTION ICI : YAW remis à 0.0 au lieu de 3.14159 <---
            arguments=['0.10', '0.0', '0.15', '0.0', '0.0', '0.0', 'base_link', 'laser_frame']
        ),
        Node(
            package='tf2_ros',
            executable='static_transform_publisher',
            name='base_to_imu',
            arguments=['0.0', '0.0', '0.05', '0.0', '0.0', '0.0', 'base_link', 'imu_link']
        ),
        
        # 2. L'Odométrie Laser brute (RF2O)
        Node(
            package='rf2o_laser_odometry',
            executable='rf2o_laser_odometry_node',
            name='rf2o_laser_odometry',
            output='screen',
            parameters=[{
                'laser_scan_topic': '/scan',   
                'odom_topic': '/odom_rf2o',    
                'publish_tf': False,           # L'EKF gérera le TF, pas Rf2o !
                'base_frame_id': 'base_link',
                'odom_frame_id': 'odom',
                'init_pose_from_topic': '',
                'freq': 20.0
            }]
        ),

        # 3. Le Filtre EKF (Fusion Lidar Odom + IMU)
        Node(
            package='robot_localization',
            executable='ekf_node',
            name='ekf_filter_node',
            output='screen',
            parameters=[os.path.join(pkg_share, 'config', 'ekf.yaml')],
            remappings=[('/odometry/filtered', '/odom')]
        ),

        # 4. Le SLAM Toolbox
        Node(
            package='slam_toolbox',
            executable='async_slam_toolbox_node',
            name='slam_toolbox',
            output='screen',
            parameters=[{
                'use_sim_time': False,
                'mode': 'mapping',
                'map_frame': 'map',
                'base_frame': 'base_link',
                'odom_frame': 'odom',
                'scan_topic': '/scan', 
                
                # --- RÉGLAGES GLOBAUX ---
                'resolution': 0.05,
                'max_laser_range': 12.0,
                'transform_timeout': 3.0,
                'tf_buffer_duration': 10.0,
                'scan_queue_size': 20,
                
                # --- RÉGLAGES ANTI-GHOSTING ---
                'map_update_interval': 2.0,        
                'minimum_time_interval': 0.5,      
                'minimum_travel_distance': 0.25,   
                'minimum_travel_heading': 0.3,     
                'use_scan_matching': True,
                'use_scan_barycenter': True,       
                'do_loop_closing': True
            }]
        )
    ])
